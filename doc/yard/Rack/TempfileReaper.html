def call(env)
  env[RACK_TEMPFILES] ||= []

  begin
    _, _, body = response = @app.call(env)
  rescue Exception
    env[RACK_TEMPFILES]&.each(&:close!)
    raise
  end

  response[2] = BodyProxy.new(body) do
    env[RACK_TEMPFILES]&.each(&:close!)
  end

  response
end
